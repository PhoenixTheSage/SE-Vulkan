set(shader_outputs)
foreach(shader spot_depth_copy_vs spot_depth_copy_ps sprite_vs sprite_ps mesh_vs mesh_ps shadow_vs shadow_ps world_compose_ps world_highlight_compose_ps world_post_compose_ps world_shadow_ps voxel_copy_ps flare_occlusion_cs ray_query_cs display_upscale_cs display_interpolate_cs gpu_opaque_reference_cs gpu_opaque_reference_vs gpu_opaque_reference_ps gpu_opaque_reference_depth_ps)
  set(extra_flags "")
  set(val_env vulkan1.4)
  if(shader MATCHES "_vs$")
    set(profile vs_6_0)
  elseif(shader STREQUAL "ray_query_cs")
    set(profile cs_6_5)
    set(extra_flags "-fspv-extension=SPV_KHR_ray_query")
    set(val_env vulkan1.4)
  elseif(shader MATCHES "_cs$")
    set(profile cs_6_0)
  else()
    set(profile ps_6_0)
  endif()
  set(output "${CMAKE_BINARY_DIR}/shaders/${shader}.spv")
  if(shader MATCHES "^gpu_opaque_reference_")
    set(shader_layout -fvk-use-scalar-layout)
  else()
    set(shader_layout -fvk-use-dx-layout)
  endif()
  set(dxc_cmd "${SEVK_DXC}" -spirv -fspv-target-env=vulkan1.3 ${shader_layout} -Zpr -T ${profile} -E main)
  if(extra_flags)
    list(APPEND dxc_cmd ${extra_flags})
  endif()
  list(APPEND dxc_cmd -Fo "${output}" "${CMAKE_CURRENT_SOURCE_DIR}/../Assets/Shaders/${shader}.hlsl")
  set(shader_includes)
  if(shader MATCHES "^gpu_opaque_reference_")
    list(APPEND shader_includes "${CMAKE_CURRENT_SOURCE_DIR}/../Assets/Shaders/gpu_opaque_reference.hlsli")
  endif()
  add_custom_command(OUTPUT "${output}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shaders"
    COMMAND ${dxc_cmd}
    COMMAND "${sevk_validator}" --target-env ${val_env} --scalar-block-layout "${output}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../Assets/Shaders/${shader}.hlsl" ${shader_includes} ${sevk_validator_dependency} VERBATIM)
  list(APPEND shader_outputs "${output}")
endforeach()
